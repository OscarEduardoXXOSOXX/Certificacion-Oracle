# ¿Qué es Oracle?

# 

# Oracle Database es un Sistema de Gestión de Base de Datos Relacional (RDBMS) desarrollado por Oracle Corporation. Es un software diseñado para almacenar, gestionar y recuperar grandes volúmenes de datos de forma eficiente.

# &nbsp;   • Modelo: Se basa en el modelo relacional (tablas relacionadas por llaves), aunque hoy es un sistema multimodelo, capaz de manejar documentos JSON, grafos, y datos espaciales.

# &nbsp;   • Naturaleza: Es multiplataforma y destaca por su arquitectura Multitenant, que permite gestionar muchas bases de datos dentro de una sola instancia (CDB/PDB).

# &nbsp;     

# Importancia de Oracle

# Oracle sigue siendo el líder indiscutible para aplicaciones críticas por:

# &nbsp;   • Escalabilidad: Capacidad de manejar desde pequeñas aplicaciones hasta bases de datos de petabytes.

# &nbsp;   • Seguridad: Implementa los estándares más altos de cifrado y control de acceso.

# &nbsp;   • Disponibilidad: Con tecnologías como Real Application Clusters (RAC), permite que las bases de datos nunca se detengan, incluso si falla un servidor.

# &nbsp;   • Mercado: Es el estándar en el sector bancario, gubernamental y de telecomunicaciones.

# 

# 

# Ventajas de Oracle

# &nbsp;   • Gestión Autónoma: Incluye capacidades de "Self-Driving" (Oracle Autonomous Database) que se auto-parchea y auto-optimiza mediante IA.

# &nbsp;   • Portabilidad: Se puede ejecutar de forma local (on-premise), en la nube (Oracle Cloud Infrastructure) o en entornos híbridos.

# &nbsp;   • Control de Concurrencia: Excelente manejo de múltiples usuarios accediendo a los mismos datos simultáneamente sin bloquearse.

# &nbsp;   • PL/SQL: Un lenguaje procedimental potente que permite programar lógica compleja directamente en la base de datos.

# 

# Fundamentos de SQL en Oracle

# SQL (Structured Query Language) es el lenguaje estándar para interactuar con Oracle(DML, DDL, DCL, TCL). Sus fundamentos se basan en:

# &nbsp;   • Esquemas: Cada usuario tiene un "esquema" que contiene sus objetos (tablas, índices, etc.).

# &nbsp;   • Integridad de datos: Uso de restricciones (Constraints) como PRIMARY KEY, FOREIGN KEY, UNIQUE y NOT NULL.

# &nbsp;   • No distinción de mayúsculas: SQL no distingue entre mayúsculas y minúsculas en comandos (aunque es buena práctica usarlas), pero sí en los datos almacenados.

# 

# 

# Comandos comunes y su sintaxis

# Aquí los comandos esenciales para empezar:

# &nbsp;   • Crear una tabla:

# &nbsp;     CREATE TABLE empleados (

# &nbsp;         id\_emp NUMBER(5) PRIMARY KEY,

# &nbsp;         nombre VARCHAR2(50),

# &nbsp;         fecha\_ingreso DATE

# &nbsp;     );

# 

# &nbsp;   • Insertar datos:

# &nbsp;     INSERT INTO empleados (id\_emp, nombre) VALUES (1, 'Juan Perez');

# &nbsp;     

# &nbsp;   • Consultar datos (con filtro):

# &nbsp;     SELECT nombre FROM empleados WHERE id\_emp = 1;

# &nbsp;     

# &nbsp;   • Actualizar datos:

# &nbsp;     UPDATE empleados SET nombre = 'Juan G. Perez' WHERE id\_emp = 1;

# 

# SQL: DML y DDL

# El lenguaje SQL se divide en subgrupos según su función:

# DML (Data Manipulation Language)

# Se usa para gestionar los datos dentro de las tablas. Los cambios requieren un COMMIT para ser permanentes.

# &nbsp;   • SELECT: Recuperar datos.

# &nbsp;   • INSERT: Añadir filas.

# &nbsp;   • UPDATE: Modificar filas existentes.

# &nbsp;   • DELETE: Eliminar filas.

# 

# 

# DDL (Data Definition Language)

# Se usa para definir la estructura de la base de datos (objetos). Estos comandos ejecutan un commit implícito.

# &nbsp;   • CREATE: Crear objetos (tablas, vistas, índices).

# &nbsp;   • ALTER: Modificar la estructura de un objeto existente (ej. añadir una columna).

# &nbsp;   • DROP: Eliminar un objeto permanentemente.

# &nbsp;   • TRUNCATE: Vaciar el contenido de una tabla pero mantener su estructura.
